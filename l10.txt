Create a standard ESP32 project in Wokwi. Add a DHT22.
Wire DHT22 VCC → ESP32 3V3, DATA → GPIO 15, GND → GND.
Install `DHT sensor library` and `PubSubClient`.
Use these core definitions in the ESP32 firmware:
#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"
#define DHTPIN 15
#define DHTTYPE DHT22
const char* WIFI_SSID = "Wokwi-GUEST";
const char* MQTT_SERVER = "test.mosquitto.org";
const int MQTT_PORT = 1883;
const char* TEMP_TOPIC = "smartfactory-lab-2026/team01/machine01/temperature";
const char* HUM_TOPIC = "smartfactory-lab-2026/team01/machine01/humidity";
const char* STATUS_TOPIC = "smartfactory-lab-2026/team01/machine01/status";

// ADDED: Required object declarations
DHT dht(DHTPIN, DHTTYPE);
WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient);

// ADDED: Required Arduino initialization function
void setup() {
  Serial.begin(115200);  // ADDED: Display messages in Serial Monitor

  // Your original initialization code
  dht.begin();
  WiFi.begin(WIFI_SSID, "");
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("WiFi connected!");
  Serial.println(WiFi.localIP());

  // Your original MQTT connection code
  mqttClient.setServer(MQTT_SERVER, MQTT_PORT);
  while (!mqttClient.connected()) {
    if (mqttClient.connect("ESP32-MACHINE-01")) {
      mqttClient.publish(STATUS_TOPIC, "ONLINE");
    } else {
      delay(2000);
    }
  }
}

// ADDED: Required Arduino repeating function
void loop() {
  // Your original sensor-reading and publishing code
  mqttClient.loop();
  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();
  char tempString[8], humString[8];
  dtostrf(temperature,1,2,tempString);
  dtostrf(humidity,1,2,humString);
  mqttClient.publish(TEMP_TOPIC,tempString);
  mqttClient.publish(HUM_TOPIC,humString);
  delay(2000);
}
